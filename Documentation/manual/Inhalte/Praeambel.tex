% Assembly step macro
\newcommand{\assemblystep}[4]{%
  \noindent
  \begin{tikzpicture}[remember picture]
    % Outer box
    \draw (0,0) rectangle (16,-10);

    % Step number
    \node[anchor=north west] at (0,-0.2) {\textbf{Step #1}};

    % Tools section
    \node[anchor=north west] at (10.3,0.1) {#3};
    \draw (10,0) -- (10,-1) -- (16,-1);

    % Image area
    \draw (0,-1) -- (10,-1) -- (10,-10);
    \node at (5,-5.5) {\includegraphics[width=9cm]{#2}};

    % Parts and instructions node
    \node[anchor=north west, text width=5.5cm, align=left] at (10.3,-1) {\small{#4}};
  \end{tikzpicture}
}
